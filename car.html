    function setupMediaSession(title, artist, artworkUrl) {
        if ('mediaSession' in navigator) {
            navigator.mediaSession.metadata = new MediaMetadata({
                title: title,
                artist: artist,
                album: "Rahul's Music Library",
                artwork: [
                    { src: artworkUrl || 'https://images.unsplash.com/photo-1511671782779-c97d3d27a1d4?w=512&h=512&fit=crop', sizes: '512x512', type: 'image/jpeg' }
                ]
            });

            // Steering wheel & earphone control actions
            navigator.mediaSession.setActionHandler('play', () => {
                if (player && player.playVideo) player.playVideo();
                startBackgroundAudio();
            });
            navigator.mediaSession.setActionHandler('pause', () => {
                if (player && player.pauseVideo) player.pauseVideo();
                pauseBackgroundAudio();
            });
            navigator.mediaSession.setActionHandler('previoustrack', playPrevTrack);
            navigator.mediaSession.setActionHandler('nexttrack', playNextTrack);
            navigator.mediaSession.setActionHandler('seekto', (details) => {
                if (player && player.seekTo && details.seekTime) {
                    player.seekTo(details.seekTime, true);
                }
            });
            navigator.mediaSession.setActionHandler('seekbackward', (details) => {
                if (!player || !player.getCurrentTime) return;
                const skip = details.seekOffset || 10;
                player.seekTo(Math.max(player.getCurrentTime() - skip, 0), true);
            });
            navigator.mediaSession.setActionHandler('seekforward', (details) => {
                if (!player || !player.getCurrentTime) return;
                const skip = details.seekOffset || 10;
                player.seekTo(player.getCurrentTime() + skip, true);
            });
        }
    }

    // --- 8. Audio Visualizer (Only runs in Audio Mode) ---
    const visualizerCanvas = document.getElementById('visualizer-canvas');
    const vCtx = visualizerCanvas.getContext('2d');
    let isVisualizerRunning = false;
    let visualizerAnimFrame;

    function resizeCanvas() {
        visualizerCanvas.width = visualizerCanvas.offsetWidth;
        visualizerCanvas.height = visualizerCanvas.offsetHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    setTimeout(resizeCanvas, 500);

    function startVisualizer() {
        if (isVisualizerRunning) return;
        isVisualizerRunning = true;
        let phase = 0;

        function renderFrame() {
            // Stop rendering if not running or if we left audio mode
            if (!isVisualizerRunning || !videoWrapper.classList.contains('audio-only')) {
                isVisualizerRunning = false;
                return;
            }
            vCtx.clearRect(0, 0, visualizerCanvas.width, visualizerCanvas.height);
            
            const accent = getComputedStyle(document.body).getPropertyValue('--accent-color').trim() || '#00d2ff';
            const bars = 48;
            const barWidth = visualizerCanvas.width / bars;
            phase += 0.08;

            for (let i = 0; i < bars; i++) {
                // Organic frequency wave simulation synced with audio rhythm
                const wave1 = Math.sin(phase + i * 0.22);
                const wave2 = Math.cos(phase * 0.7 + i * 0.15);
                const combined = Math.abs(wave1 * 0.6 + wave2 * 0.4);
                const barHeight = Math.max(6, combined * (visualizerCanvas.height * 0.85));

                const x = i * barWidth;
                const y = visualizerCanvas.height - barHeight;

                const grad = vCtx.createLinearGradient(0, y, 0, visualizerCanvas.height);
                grad.addColorStop(0, accent);
                grad.addColorStop(1, 'rgba(0,0,0,0.05)');

                vCtx.fillStyle = grad;
                vCtx.fillRect(x + 2, y, barWidth - 4, barHeight);
            }

            visualizerAnimFrame = requestAnimationFrame(renderFrame);
        }
        renderFrame();
    }

    function stopVisualizer() {
        isVisualizerRunning = false;
        cancelAnimationFrame(visualizerAnimFrame);
        vCtx.clearRect(0, 0, visualizerCanvas.width, visualizerCanvas.height);
    }

    // --- 9. Player Controls Setup ---
    function setupControls() {
        const playBtn = document.getElementById('play-pause-btn');
        const muteBtn = document.getElementById('mute-btn');
        const seekBar = document.getElementById('seek-bar');
        const volumeBar = document.getElementById('volume-bar');
        const nextBtn = document.getElementById('next-btn');
        const prevBtn = document.getElementById('prev-btn');

        nextBtn.addEventListener('click', playNextTrack);
        prevBtn.addEventListener('click', playPrevTrack);

        const toggleVideoBtn = document.getElementById('toggle-video-btn');
        const iconCamera = document.getElementById('icon-camera');
        const iconHeadphone = document.getElementById('icon-headphone');
        let isAudioOnly = false;

        toggleVideoBtn.addEventListener('click', () => {
            isAudioOnly = !isAudioOnly;
            if(isAudioOnly) {
                videoWrapper.classList.add('audio-only');
                iconCamera.style.display = 'none';
                iconHeadphone.style.display = 'block';
                toggleVideoBtn.style.color = "var(--accent-color)";
                
                // If song is currently playing, start visualizer immediately
                if (player && player.getPlayerState && player.getPlayerState() === YT.PlayerState.PLAYING) {
                    startVisualizer();
                }
            } else {
                videoWrapper.classList.remove('audio-only');
                iconCamera.style.display = 'block';
                iconHeadphone.style.display = 'none';
                toggleVideoBtn.style.color = "#fff";
                
                // Stop visualizer when switching back to video mode
                stopVisualizer();
            }
        });

        playBtn.addEventListener('click', () => {
            if(!player || !player.getPlayerState) return;
            if (player.getPlayerState() === YT.PlayerState.PLAYING) {
                player.pauseVideo();
            } else {
                player.playVideo();
                startBackgroundAudio();
            }
        });

        // Custom Volume Slider
        function updateVolumeBarColor(val) {
            volumeBar.style.background = `linear-gradient(to right, var(--accent-color) ${val}%, rgba(255, 255, 255, 0.25) ${val}%)`;
        }
        updateVolumeBarColor(100);

        volumeBar.addEventListener('input', (e) => {
            const vol = e.target.value;
            if (player && player.setVolume) {
                player.setVolume(vol);
                if (vol > 0 && player.isMuted && player.isMuted()) player.unMute();
            }
            updateVolumeBarColor(vol);
            muteBtn.style.opacity = vol == 0 ? '0.5' : '1';
        });

        muteBtn.addEventListener('click', () => {
            if (!player || !player.isMuted) return;
            if(player.isMuted()) {
                player.unMute();
                muteBtn.style.opacity = '1';
                volumeBar.value = player.getVolume() || 100;
            } else {
                player.mute();
                muteBtn.style.opacity = '0.5';
                volumeBar.value = 0;
            }
            updateVolumeBarColor(volumeBar.value);
        });

        seekBar.addEventListener('input', (e) => {
            if(!player || !player.getDuration) return;
            player.seekTo((e.target.value / 100) * player.getDuration(), true);
            updateSeekBarColor(e.target.value);
        });

        document.getElementById('fullscreen-btn').addEventListener('click', () => {
            !document.fullscreenElement ? videoWrapper.requestFullscreen() : document.exitFullscreen();
        });
    }

    function updateSeekBarColor(percent) {
        document.getElementById('seek-bar').style.background = `linear-gradient(to right, var(--accent-color) ${percent}%, rgba(255, 255, 255, 0.25) ${percent}%)`;
    }

    function onPlayerStateChange(event) {
        const playIcon = document.getElementById('play-icon');
        const pauseIcon = document.getElementById('pause-icon');

        if (event.data === YT.PlayerState.PLAYING) {
            playIcon.style.display = 'none';
            pauseIcon.style.display = 'block';
            timeUpdater = setInterval(updateUI, 500);
            
            // Only start Visualizer if in Audio Mode
            if (videoWrapper.classList.contains('audio-only')) {
                startVisualizer();
            }
            startBackgroundAudio();
            if ('mediaSession' in navigator) navigator.mediaSession.playbackState = 'playing';
        } 
        else if (event.data === YT.PlayerState.ENDED) {
            clearInterval(timeUpdater);
            stopVisualizer();
            playNextTrack();
        }
        else {
            playIcon.style.display = 'block';
            pauseIcon.style.display = 'none';
            clearInterval(timeUpdater);
            stopVisualizer();
            pauseBackgroundAudio();
            if ('mediaSession' in navigator) navigator.mediaSession.playbackState = 'paused';
        }
    }

    function updateUI() {
        if (!player || !player.getDuration) return;
        const dur = player.getDuration(), cur = player.getCurrentTime();
        if (dur > 0) {
            const percent = (cur / dur) * 100;
            document.getElementById('seek-bar').value = percent;
            updateSeekBarColor(percent);
            document.getElementById('current-time').innerText = fmt(cur);
            document.getElementById('duration-time').innerText = fmt(dur);

            // Update MediaSession Position State for lockscreens
            if ('mediaSession' in navigator && 'setPositionState' in navigator.mediaSession) {
                try {
                    navigator.mediaSession.setPositionState({
                        duration: dur,
                        playbackRate: player.getPlaybackRate ? player.getPlaybackRate() : 1,
                        position: cur
                    });
                } catch(e) {}
            }
        }
    }

    function fmt(time) {
        if(isNaN(time)) return "0:00";
        const m = Math.floor(time / 60), s = Math.floor(time % 60);
        return `${m}:${s < 10 ? '0' : ''}${s}`;
    }
